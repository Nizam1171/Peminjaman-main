dalam folder backend
   cp .env.example .env		= copy file .env
edit file .env

kembali ke folder ~/Peminjaman-main
   docker compose up -d

install composer
   docker compose run -rm app-laravel composer install

masuk folder backend
cek kepemilikan composer tersedia di vendor
   ls -la vendor/autoload.php

masih dalam folder backend, jalankan
   docker compose run -rm app-laravel ls -la /var/www/vendor/autoload.php

cek list container
   docker compose ps -a

cek log terakhir
   docker compose logs --tail=100 app-laravel

install dengan artisan
   docker compose run -rm app-laravel php artisan --version
